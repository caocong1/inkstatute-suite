       ```python
       def filter_artifact_metadata(metadata: JsonObject) -> JsonObject:
           """Filter artifact metadata to allowed keys only, strip backend noise."""
           filtered: JsonObject = {}
           for key, value in metadata.items():
               if str(key).lower() not in _ALLOWED_ARTIFACT_METADATA_KEYS:
                   continue
               filtered[key] = _sanitize_artifact_value(value)
           return filtered


       def _sanitize_artifact_value(value: Any) -> Any:
           if isinstance(value, dict):
               return {
                   k: _sanitize_artifact_value(v)
                   for k, v in value.items()
                   if str(k).lower() in _ALLOWED_ARTIFACT_METADATA_KEYS
               }
           if isinstance(value, list):
               return [_sanitize_artifact_value(item) for item in value]
           if isinstance(value, str):
               return _sanitize_artifact_text(value)
           return value


       def _sanitize_artifact_text(text: str) -> str:
           """Strip URLs, token-like patterns, and model name patterns from artifact text."""
           sanitized = _URL_RE.sub("[url_redacted]", text) if _URL_RE.search(text) else text
           sanitized = _TOKEN_RE.sub("token=[redacted]", sanitized) if _TOKEN_RE.search(sanitized) else sanitized
           return sanitized
       ```
       Note: `_URL_RE` and `_TOKEN_RE` need to be defined before this function. Add them after the `_HARD_BLOCK_RULES` section or reuse existing patterns. Actually, the output_boundary.py doesn't currently have `_URL_RE` or `_TOKEN_RE` — those are in the adapter files. Add simple regex patterns:

       And use those in `_sanitize_artifact_text`.
