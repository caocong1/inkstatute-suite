### 1. Create `packages/solution-loader/src/inkstatute_solution_loader/privacy_scan.py`

```python
from pathlib import Path
from typing import Iterable
from platform_runtime import Issue
# reuse the sanitize module via a known import path
from importlib import import_module
_lawfirm_sanitize = None
def _get_lawfirm_sanitize():
    global _lawfirm_sanitize
    if _lawfirm_sanitize is None:
        try:
            _lawfirm_sanitize = import_module("inkstatute_solutions_lawfirm.sanitize")
        except ImportError:
            _lawfirm_sanitize = False
    return _lawfirm_sanitize or None

TEXT_EXTENSIONS = {".md", ".txt", ".json", ".template", ".jsonl"}
BINARY_REFUSE_EXTENSIONS = {".log", ".db", ".sqlite", ".sqlite3", ".png", ".jpg", ".jpeg", ".gif", ".mp3", ".mp4", ".wav", ".pdf", ".docx", ".doc"}

def scan_bundle_dir(bundle_dir: Path) -> list[Issue]:
    issues: list[Issue] = []
    if not bundle_dir.exists():
        return [Issue(path=str(bundle_dir), rule="bundle_dir_missing", message="bundle directory does not exist", severity="error")]
    for path in sorted(bundle_dir.rglob("*")):
        if not path.is_file():
            continue
        rel = str(path.relative_to(bundle_dir))
        # Binary/file-name refusal
        if path.suffix.lower() in BINARY_REFUSE_EXTENSIONS:
            issues.append(Issue(path=rel, rule="binary_or_sensitive_extension", message=f"refused binary/sensitive file type: {path.suffix}", severity="error"))
            continue
        # Text file content scan
        if path.suffix.lower() in TEXT_EXTENSIONS:
            try:
                text = path.read_text(encoding="utf-8", errors="replace")
            except OSError:
                continue
            sanitize = _get_lawfirm_sanitize()
            if sanitize is not None:
                for rule in sanitize.scan_text(text):
                    issues.append(Issue(path=rel, rule=rule, message=f"text matches refusal pattern: {rule}", severity="error"))
    return issues
```

The lazy import ensures the privacy scanner is usable even if `inkstatute_solutions_lawfirm` is not on PYTHONPATH (test-time fallback to name-only scan).
