# Plan 02: Token-Based Auth with Role+Tenant Claims and Static-Token Fallback

## Objective

Implement short-lived HMAC-SHA256 signed tokens carrying `role` and `tenant_id` claims, a `resolve_identity()` function that extracts claims from a Bearer token when `rbac_enabled=True` and falls back to the v1.6 `check_auth_header` static-token path when `rbac_enabled=False`. This satisfies RBAC-01 (token claims) and preserves the static-token fallback (success criterion 3 and 4).

       ```python
       from __future__ import annotations
       import base64, hashlib, hmac, json, time
       from dataclasses import dataclass
       from .roles import Role

       DEFAULT_TOKEN_TTL_SECONDS = 900

       class RbacTokenError(Exception):
           """Raised when an RBAC token is invalid, tampered, or expired."""

       @dataclass(frozen=True, slots=True)
       class RbacTokenClaims:
           role: str
           tenant_id: str
           iat: int
           exp: int

       def _b64url_encode(data: bytes) -> str:
           return base64.urlsafe_b64encode(data).rstrip(b"=").decode("ascii")

       def _b64url_decode(text: str) -> bytes:
           padding = "=" * (-len(text) % 4)
           return base64.urlsafe_b64decode(text + padding)

       def issue_token(secret, *, role, tenant_id, ttl_seconds=DEFAULT_TOKEN_TTL_SECONDS, now=None):
           iat = int(now if now is not None else time.time())
           exp = iat + int(ttl_seconds)
           payload = {"role": role.value if isinstance(role, Role) else str(role), "tenant_id": tenant_id, "iat": iat, "exp": exp}
           payload_json = json.dumps(payload, separators=(",", ":"), sort_keys=True)
           payload_b64 = _b64url_encode(payload_json.encode("utf-8"))
           sig = hmac.new(secret.encode("utf-8"), payload_b64.encode("ascii"), hashlib.sha256).digest()
           sig_b64 = _b64url_encode(sig)
           return f"{payload_b64}.{sig_b64}"

       def verify_token(token, secret, now=None):
           now = int(now if now is not None else time.time())
           if not isinstance(token, str) or token.count(".") != 1:
               raise RbacTokenError("invalid token format")
           payload_b64, sig_b64 = token.split(".")
           try:
               payload_json = _b64url_decode(payload_b64).decode("utf-8")
               provided_sig = _b64url_decode(sig_b64)
           except Exception as exc:
               raise RbacTokenError("invalid token encoding") from exc
           expected_sig = hmac.new(secret.encode("utf-8"), payload_b64.encode("ascii"), hashlib.sha256).digest()
           if not hmac.compare_digest(provided_sig, expected_sig):
               raise RbacTokenError("invalid signature")
           try:
               payload = json.loads(payload_json)
           except Exception as exc:
               raise RbacTokenError("invalid payload json") from exc
           exp = int(payload.get("exp", 0))
           if exp <= now:
               raise RbacTokenError("token expired")
           return RbacTokenClaims(role=str(payload["role"]), tenant_id=str(payload["tenant_id"]), iat=int(payload["iat"]), exp=exp)

       __all__ = ["DEFAULT_TOKEN_TTL_SECONDS", "RbacTokenClaims", "RbacTokenError", "issue_token", "verify_token"]
       ```

    1. Create `packages/rbac/src/rbac/identity.py`:
       ```python
       from __future__ import annotations
       from dataclasses import dataclass
       from .roles import Role
       from .token import RbacTokenError, verify_token

       _BEARER_PREFIX = "Bearer "

       @dataclass(frozen=True, slots=True)
       class Identity:
           role: Role
           tenant_id: str

       def _strip_bearer(header_value: str | None) -> str | None:
           if header_value is None or not header_value.startswith(_BEARER_PREFIX):
               return None
           return header_value[len(_BEARER_PREFIX):]

       def resolve_identity(auth_header, *, rbac_secret=None, static_token=None, rbac_enabled=False, now=None):
           if rbac_enabled:
               token = _strip_bearer(auth_header)
               if token is None or rbac_secret is None:
                   return None
               try:
                   claims = verify_token(token, rbac_secret, now=now)
               except RbacTokenError:
                   return None
               try:
                   role = Role(claims.role)
               except ValueError:
                   return None
               return Identity(role=role, tenant_id=claims.tenant_id)
           # static-token fallback (v1.6 behavior)
           from admin_console_api.auth import check_auth_header  # lazy import
           if check_auth_header(auth_header, static_token):
               return Identity(role=Role.ADMIN, tenant_id="default")
           return None

       __all__ = ["Identity", "resolve_identity"]
       ```
