    1. Create `scripts/scan_rbac_annotations.py`:
       ```python
       #!/usr/bin/env python3
       """CI scan: fail when any admin API endpoint lacks an RBAC role annotation."""
       from __future__ import annotations
       import sys
       from typing import Iterable, Mapping

       def find_unannotated(route_prefixes: Iterable[str], matrix: Mapping[str, object]) -> list[str]:
           return [p for p in route_prefixes if p not in matrix]

       def main(argv: list[str] | None = None) -> int:
           from admin_console_api.routes import ROUTE_PREFIXES
           from rbac.matrix import PERMISSIONS_MATRIX
           unannotated = find_unannotated(ROUTE_PREFIXES, PERMISSIONS_MATRIX)
           if unannotated:
               print(f"FAIL: unannotated endpoints (no RBAC role): {unannotated}", file=sys.stderr)
               return 1
           print(f"OK: all {len(ROUTE_PREFIXES)} admin API endpoints have RBAC annotations")
           return 0

       if __name__ == "__main__":
           raise SystemExit(main())
       ```
